cmake_minimum_required(VERSION 3.16)
# Migration reference only. Production builds use rust/ipa_graph and the
# Android CMake project links that Rust static library, not this implementation.
project(ipa_graph_reference VERSION 1.0 LANGUAGES CXX)

add_library(ipa_graph STATIC src/ipa_graph.cpp)
target_include_directories(ipa_graph PUBLIC include)
target_compile_features(ipa_graph PUBLIC cxx_std_17)
set_target_properties(ipa_graph PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    target_compile_options(ipa_graph PRIVATE /W4)
else()
    target_compile_options(ipa_graph PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(NOT ANDROID)
    add_executable(ipa_graph_cli src/ipa_graph_cli.cpp)
    target_link_libraries(ipa_graph_cli PRIVATE ipa_graph)
    set_target_properties(ipa_graph_cli PROPERTIES OUTPUT_NAME ipa_graph)
    include(CTest)
    if(BUILD_TESTING)
        find_package(Python3 COMPONENTS Interpreter REQUIRED)
        find_package(Threads REQUIRED)
        add_executable(ipa_graph_test tests/runtime_test.cpp)
        target_link_libraries(ipa_graph_test PRIVATE ipa_graph Threads::Threads)
        add_test(NAME graph_fixture COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/make_fixture.py ${CMAKE_CURRENT_BINARY_DIR}/fixture.ipadawg)
        set_tests_properties(graph_fixture PROPERTIES FIXTURES_SETUP tiny_graph)
        add_test(NAME graph_runtime COMMAND ipa_graph_test ${CMAKE_CURRENT_BINARY_DIR}/fixture.ipadawg)
        set_tests_properties(graph_runtime PROPERTIES FIXTURES_REQUIRED tiny_graph)
        add_test(NAME graph_fixture32 COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/make_fixture.py ${CMAKE_CURRENT_BINARY_DIR}/fixture32.ipadawg 32)
        set_tests_properties(graph_fixture32 PROPERTIES FIXTURES_SETUP tiny_graph32)
        add_test(NAME graph_runtime32 COMMAND ipa_graph_test ${CMAKE_CURRENT_BINARY_DIR}/fixture32.ipadawg)
        set_tests_properties(graph_runtime32 PROPERTIES FIXTURES_REQUIRED tiny_graph32)
    endif()
endif()
