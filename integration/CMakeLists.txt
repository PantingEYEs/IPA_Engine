cmake_minimum_required(VERSION 3.25)
project(ipa_board_ort_extensions LANGUAGES C CXX)
if(NOT ANDROID)
  message(FATAL_ERROR "This profile only builds Android libraries")
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(OCOS_BUILD_PRESET "" CACHE STRING "" FORCE)
set(OCOS_USE_CUDA OFF CACHE BOOL "" FORCE)
set(OCOS_ENABLE_PYTHON OFF CACHE BOOL "" FORCE)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_SHARED_LINKER_FLAGS "${CMAKE_SHARED_LINKER_FLAGS} -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384")
foreach(op TF_STRING RE2_REGEX GPT2_TOKENIZER TRIE_TOKENIZER WORDPIECE_TOKENIZER BERT_TOKENIZER BLINGFIRE MATH DLIB OPENCV_CODECS CV2 VISION AUDIO AZURE VENDOR_IMAGE_CODECS)
  set(OCOS_ENABLE_${op} OFF CACHE BOOL "Disabled in E5 profile" FORCE)
endforeach()
set(OCOS_ENABLE_SPM_TOKENIZER ON CACHE BOOL "Required by pinned XLM-R tokenizer" FORCE)
set(OCOS_ENABLE_CPP_EXCEPTIONS ON CACHE BOOL "SentencePiece requires exceptions" FORCE)
foreach(opt CTEST C_API STATIC_LIB SELECTED_OPLIST)
  set(OCOS_ENABLE_${opt} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(opt PYTHON JAVA ANDROID)
  set(OCOS_BUILD_${opt} OFF CACHE BOOL "Built without upstream Gradle packaging" FORCE)
endforeach()
set(OCOS_BUILD_SHARED_LIB ON CACHE BOOL "" FORCE)
set(OCOS_ONNXRUNTIME_VERSION "1.23.2" CACHE STRING "" FORCE)
set(ONNXRUNTIME_INCLUDE_DIR "${CMAKE_CURRENT_LIST_DIR}/onnxruntime-1.23.2-headers")
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "Pinned dependencies supplied by prepare.py" FORCE)
foreach(dep PROTOBUF SPM GSL)
  string(TOLOWER "${dep}" lower)
  set(FETCHCONTENT_SOURCE_DIR_${dep} "${ORTX_DEPS}/${lower}" CACHE PATH "" FORCE)
endforeach()
add_subdirectory("${ORTX_SOURCE}" upstream)
# Same JNI bridge/static operator link as v0.13.0 Android packaging, with
# javac-generated headers and no Gradle formatting side effects.
add_library(onnxruntime_extensions4j_jni SHARED
  "${ORTX_SOURCE}/java/src/main/native/ai_onnxruntime_extensions_OrtxLibrary.c")
target_include_directories(onnxruntime_extensions4j_jni PRIVATE
  "${ORTX_JAVA_HEADERS}" "${ORTX_SOURCE}/include")
target_link_libraries(onnxruntime_extensions4j_jni PRIVATE ortcustomops)
target_link_options(onnxruntime_extensions4j_jni PRIVATE
  "-Wl,--version-script=${ORTX_SOURCE}/java/ortx_jni.ver")
set_target_properties(onnxruntime_extensions4j_jni PROPERTIES
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")

foreach(target extensions_shared onnxruntime_extensions4j_jni)
  target_link_options(${target} PRIVATE "-Wl,-Map=${CMAKE_BINARY_DIR}/${target}.map")
endforeach()
